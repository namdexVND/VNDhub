-- ====================================================================
-- 🚀 VND HUB | ULTRA PRO FARM - FULL MENU & LOGIC
-- ====================================================================
local RunService = game:GetService("RunService")
local VirtualUser = game:GetService("VirtualUser")
local LocalPlayer = game:GetService("Players").LocalPlayer
local Workspace = game:GetService("Workspace")

local settings = {
    Enabled = false,
    TargetName = "CityNPC",
    DashDistance = 15,
    AttackRange = 25
}

-- [ GIAO DIỆN MENU ]
local ScreenGui = Instance.new("ScreenGui", LocalPlayer:WaitForChild("PlayerGui"))
local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 280, 0, 200)
MainFrame.Position = UDim2.new(0.5, -140, 0.5, -100)
MainFrame.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
MainFrame.Active = true
MainFrame.Draggable = true
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 12)
Instance.new("UIStroke", MainFrame).Color = Color3.fromRGB(100, 100, 255)

local Title = Instance.new("TextLabel", MainFrame)
Title.Text = "VND HUB | PRO FARM"
Title.Size = UDim2.new(1, 0, 0, 40)
Title.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
Title.TextColor3 = Color3.new(1,1,1)
Title.Font = Enum.Font.GothamBold
Title.Parent = MainFrame

local StartBtn = Instance.new("TextButton", MainFrame)
StartBtn.Size = UDim2.new(0.8, 0, 0, 40)
StartBtn.Position = UDim2.new(0.1, 0, 0.3, 0)
StartBtn.Text = "START FARM"
StartBtn.BackgroundColor3 = Color3.fromRGB(40, 200, 100)
StartBtn.Parent = MainFrame

local TeleBtn = Instance.new("TextButton", MainFrame)
TeleBtn.Size = UDim2.new(0.8, 0, 0, 40)
TeleBtn.Position = UDim2.new(0.1, 0, 0.65, 0)
TeleBtn.Text = "TELEPORT TO NPC"
TeleBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 255)
TeleBtn.Parent = MainFrame

StartBtn.MouseButton1Click:Connect(function()
    settings.Enabled = not settings.Enabled
    StartBtn.Text = settings.Enabled and "STOP FARM" or "START FARM"
    StartBtn.BackgroundColor3 = settings.Enabled and Color3.fromRGB(200, 40, 40) or Color3.fromRGB(40, 200, 100)
end)

TeleBtn.MouseButton1Click:Connect(function()
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") then
            LocalPlayer.Character.HumanoidRootPart.CFrame = obj.HumanoidRootPart.CFrame
        end
    end
end)

-- [ LOGIC FARM NÂNG CẤP ]
local lastDash = 0
local dashCooldown = 0.35

RunService.Heartbeat:Connect(function()
    if not settings.Enabled then return end
    
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local target = nil
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") then
            local hum = obj:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                target = obj.HumanoidRootPart
                break
            end
        end
    end

    if target then
        local dist = (hrp.Position - target.Position).Magnitude
        
        if dist > settings.AttackRange then
            hrp.CFrame = CFrame.lookAt(hrp.Position, target.Position) * CFrame.new(0, 0, -2)
        elseif tick() - lastDash > dashCooldown then
            lastDash = tick()
            local side = (math.random() > 0.5 and 1 or -1) * settings.DashDistance
            local snapPos = target.CFrame * CFrame.new(side, 0, 8)
            hrp.CFrame = CFrame.lookAt(snapPos.Position, target.Position)
            
            VirtualUser:CaptureController()
            VirtualUser:Button1Down(Vector2.new(0.5, 0.5))
            task.spawn(function()
                task.wait(0.05)
                VirtualUser:Button1Up(Vector2.new(0.5, 0.5))
            end)
        end
    end
end)
