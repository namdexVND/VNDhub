local RunService = game:GetService("RunService")
local VirtualUser = game:GetService("VirtualUser")
local LocalPlayer = game:GetService("Players").LocalPlayer

local settings = {
    Enabled = false,
    TargetName = "CityNPC", -- ĐÃ CẬP NHẬT TÊN CHÍNH XÁC
    Distance = 8,
    DashSpeed = 13.5
}

-- [ GIAO DIỆN MENU ]
local ScreenGui = Instance.new("ScreenGui", LocalPlayer:WaitForChild("PlayerGui"))
local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 280, 0, 250)
MainFrame.Position = UDim2.new(0.5, -140, 0.5, -125)
MainFrame.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
MainFrame.Active = true
MainFrame.Draggable = true
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 12)
Instance.new("UIStroke", MainFrame).Color = Color3.fromRGB(100, 100, 255)

local Title = Instance.new("TextLabel", MainFrame)
Title.Text = "VND HUB | FARM BOT"
Title.Size = UDim2.new(1, 0, 0, 50)
Title.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
Title.TextColor3 = Color3.new(1,1,1)
Title.Font = Enum.Font.GothamBold
Title.TextSize = 18
Instance.new("UICorner", Title).CornerRadius = UDim.new(0, 12)
Title.Parent = MainFrame

local StartBtn = Instance.new("TextButton", MainFrame)
StartBtn.Size = UDim2.new(0.8, 0, 0, 50)
StartBtn.Position = UDim2.new(0.1, 0, 0.3, 0)
StartBtn.Text = "START FARM"
StartBtn.BackgroundColor3 = Color3.fromRGB(40, 200, 100)
StartBtn.TextColor3 = Color3.new(1,1,1)
Instance.new("UICorner", StartBtn).CornerRadius = UDim.new(0, 8)
StartBtn.Parent = MainFrame

local TeleBtn = Instance.new("TextButton", MainFrame)
TeleBtn.Size = UDim2.new(0.8, 0, 0, 50)
TeleBtn.Position = UDim2.new(0.1, 0, 0.6, 0)
TeleBtn.Text = "TELEPORT TO NPC"
TeleBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 255)
TeleBtn.TextColor3 = Color3.new(1,1,1)
Instance.new("UICorner", TeleBtn).CornerRadius = UDim.new(0, 8)
TeleBtn.Parent = MainFrame

StartBtn.MouseButton1Click:Connect(function()
    settings.Enabled = not settings.Enabled
    StartBtn.Text = settings.Enabled and "STOP FARM" or "START FARM"
    StartBtn.BackgroundColor3 = settings.Enabled and Color3.fromRGB(200, 40, 40) or Color3.fromRGB(40, 200, 100)
end)

TeleBtn.MouseButton1Click:Connect(function()
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") then
            LocalPlayer.Character.HumanoidRootPart.CFrame = obj.HumanoidRootPart.CFrame
            break
        end
    end
end)

-- [ LOGIC PHI XUYÊN NGƯỜI (LINEAR STRAFE) ]
local direction = 1
RunService.Heartbeat:Connect(function()
    if not settings.Enabled then return end
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    local hrp = char.HumanoidRootPart
    local hum = char:FindFirstChild("Humanoid")

    local target = nil
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") and obj.Humanoid.Health > 0 then
            target = obj.HumanoidRootPart
            break
        end
    end

    if target then
        if hum then hum:ChangeState(Enum.HumanoidStateType.Physics) end
        
        -- Logic: Đổi hướng khi vượt quá biên độ 5 stud
        local relativePos = target.CFrame:PointToObjectSpace(hrp.Position)
        if math.abs(relativePos.X) > 5 then 
            direction = relativePos.X > 0 and -1 or 1 
        end
        
        -- Vị trí "Phi": Giữ khoảng cách 8 + lướt qua lại 5 stud
        local targetPos = target.Position + (target.CFrame.LookVector * 8) + (target.CFrame.RightVector * (5 * direction))
        
        local dt = RunService.Heartbeat:Wait()
        local lerpAlpha = math.clamp(settings.DashSpeed * dt, 0, 1) 
        
        hrp.CFrame = hrp.CFrame:Lerp(CFrame.new(targetPos, target.Position), lerpAlpha)

        -- Tấn công cực nhanh
        VirtualUser:CaptureController()
        VirtualUser:Button1Down(Vector2.new(0.5, 0.5))
        task.wait(0.02) 
        VirtualUser:Button1Up(Vector2.new(0.5, 0.5))
    else
        if hum then hum:ChangeState(Enum.HumanoidStateType.Running) end
    end
end)
