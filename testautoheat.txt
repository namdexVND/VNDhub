local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Player = Players.LocalPlayer

-- [ TẠO GIAO DIỆN GUI ]
-- Thay CoreGui bằng PlayerGui để tránh bị chặn ở một số game
local ScreenGui = Instance.new("ScreenGui", Player:WaitForChild("PlayerGui"))
ScreenGui.Name = "AutoBandageMenu"

local Frame = Instance.new("Frame", ScreenGui)
Frame.Size = UDim2.new(0, 200, 0, 140)
Frame.Position = UDim2.new(0.8, -100, 0.5, -70)
Frame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
Frame.Active = true
Frame.Draggable = true 
Instance.new("UICorner", Frame).CornerRadius = UDim.new(0, 8)

-- Tiêu đề
local Title = Instance.new("TextLabel", Frame)
Title.Size = UDim2.new(1, 0, 0, 30)
Title.Text = "AUTO SHOP BĂNG GẠT"
Title.TextColor3 = Color3.new(1, 1, 1)
Title.BackgroundTransparency = 1
Title.Font = Enum.Font.Bold
Title.Parent = Frame -- ĐÃ THÊM

-- Nút Bật/Tắt
local ToggleBtn = Instance.new("TextButton", Frame)
ToggleBtn.Size = UDim2.new(0.8, 0, 0, 35)
ToggleBtn.Position = UDim2.new(0.1, 0, 0.3, 0)
ToggleBtn.Text = "Auto: OFF"
ToggleBtn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
Instance.new("UICorner", ToggleBtn).CornerRadius = UDim.new(0, 4)
ToggleBtn.Parent = Frame -- ĐÃ THÊM

-- Ô nhập số lượng
local AmountBox = Instance.new("TextBox", Frame)
AmountBox.Size = UDim2.new(0.8, 0, 0, 35)
AmountBox.Position = UDim2.new(0.1, 0, 0.65, 0)
AmountBox.PlaceholderText = "Số lượng mua"
AmountBox.Text = "1"
Instance.new("UICorner", AmountBox).CornerRadius = UDim.new(0, 4)
AmountBox.Parent = Frame -- ĐÃ THÊM

-- [ LOGIC ĐIỀU KHIỂN ]
local AutoEnabled = false
local BuyAmount = 1

ToggleBtn.MouseButton1Click:Connect(function()
    AutoEnabled = not AutoEnabled
    ToggleBtn.Text = AutoEnabled and "Auto: ON" or "Auto: OFF"
    ToggleBtn.BackgroundColor3 = AutoEnabled and Color3.fromRGB(50, 200, 50) or Color3.fromRGB(200, 50, 50)
end)

AmountBox.FocusLost:Connect(function()
    BuyAmount = tonumber(AmountBox.Text) or 1
end)

-- [ LOGIC AUTO MUA ]
task.spawn(function()
    while true do
        task.wait(2)
        if AutoEnabled then
            pcall(function()
                local bandageName = "b\196\131ng g\225\186\161c"
                local hasItem = Player.Backpack:FindFirstChild(bandageName) or Player.Character:FindFirstChild(bandageName)
                
                if not hasItem then
                    local args = {
                        [1] = bandageName,
                        [2] = BuyAmount
                    }
                    ReplicatedStorage:WaitForChild("KnitPackages"):WaitForChild("_Index"):WaitForChild("sleitnick_knit@1.7.0"):WaitForChild("knit"):WaitForChild("Services"):WaitForChild("ShopService"):WaitForChild("RE"):WaitForChild("buyItem"):FireServer(unpack(args))
                end
            end)
        end
    end
end)
