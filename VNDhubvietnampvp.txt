local function getFixedKeyFor8Hours()
    local totalHours = math.floor(os.time() / 3600)
    local period = math.floor(totalHours / 8)
    math.randomseed(period)
    local randomString = ""
    for i = 1, 12 do randomString = randomString .. tostring(math.random(0, 9)) end
    return "VND_" .. randomString

local originalMaterials = {}

local function saveOriginalMaterials()
    if next(originalMaterials) == nil then
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                originalMaterials[v] = {
                    Material = v.Material,
                    Reflectance = v.Reflectance,
                    CastShadow = v.CastShadow
                }
            end
        end
    end
end

function toggleGraphicReduction(percent, enable)
    saveOriginalMaterials()

    local totalParts = 0
    for _ in pairs(originalMaterials) do
        totalParts += 1
    end

    local targetCount = math.floor(totalParts * (percent / 100))
    local currentCount = 0

    for part, data in pairs(originalMaterials) do
        if part and part.Parent then
            currentCount += 1

            if currentCount <= targetCount then
                if enable then
                    part.Material = Enum.Material.SmoothPlastic
                    part.Reflectance = 0
                    part.CastShadow = false
                else
                    part.Material = data.Material
                    part.Reflectance = data.Reflectance
                    part.CastShadow = data.CastShadow
                end
            end
        end
    end
end

local oldBrightness = Lighting.Brightness
local oldFogEnd = Lighting.FogEnd
local oldGlobalShadows = Lighting.GlobalShadows

function activateFPSBoost()

    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    end)

    Lighting.GlobalShadows = false
    Lighting.Brightness = 0
    Lighting.FogEnd = 999999

    for _, v in ipairs(workspace:GetDescendants()) do

        if v:IsA("ParticleEmitter")
        or v:IsA("Trail")
        or v:IsA("Beam")
        or v:IsA("Smoke")
        or v:IsA("Fire")
        or v:IsA("Sparkles") then

            v.Enabled = false
        end

        if v:IsA("BasePart") then
            v.CastShadow = false
        end
    end
end

function deactivateFPSBoost()

    Lighting.GlobalShadows = oldGlobalShadows
    Lighting.Brightness = oldBrightness
    Lighting.FogEnd = oldFogEnd

    for part, data in pairs(originalMaterials) do
        if part and part.Parent then
            part.Material = data.Material
            part.Reflectance = data.Reflectance
            part.CastShadow = data.CastShadow
        end
    end
end

local function getClosestPlayer()
    local closest, shortest = nil, math.huge
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hum = p.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (myRoot.Position - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest = dist; closest = p end
            end
        end
    end
    return closest
end

RunService.RenderStepped:Connect(function()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")

    -- Movement
    if _G.PvpSpeedActive and hrp and hum and hum.MoveDirection.Magnitude > 0 then hrp.CFrame += hum.MoveDirection * 0.5 end
    
    -- Spin
    if _G.PvpSpinActive and hrp then
        if not hrp:FindFirstChild("VND_Spin") then
            local s = Instance.new("BodyAngularVelocity", hrp); s.Name = "VND_Spin"; s.MaxTorque = Vector3.new(0, math.huge, 0); s.AngularVelocity = Vector3.new(0, 40, 0)
        end
    elseif hrp and hrp:FindFirstChild("VND_Spin") then hrp.VND_Spin:Destroy() end

    -- Aim
    if _G.PvpAimNoCamActive then
        local target = getClosestPlayer()
        if target and target.Character and hrp then
            hrp.CFrame = CFrame.new(hrp.Position, Vector3.new(target.Character.HumanoidRootPart.Position.X, hrp.Position.Y, target.Character.HumanoidRootPart.Position.Z))
        end
    end

    -- Hitbox
    if _G.PvpHitboxActive then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character.Humanoid.Health > 0 then
                local root = p.Character.HumanoidRootPart
                root.Size = Vector3.new(5, 5, 5); root.CanCollide = false
                if not activeHitboxes[p.Name] then
                    local box = Instance.new("BoxHandleAdornment", ScreenGui); box.Size = Vector3.new(5, 6, 5); box.Color3 = Color3.fromRGB(255, 0, 0); box.Transparency = 0.6; box.AlwaysOnTop = true; box.Adornee = root
                    activeHitboxes[p.Name] = box
                end
            end
        end
    else clearAllHitboxes() end

    -- Health
    if _G.PvpHealthActive then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= player and p.Character and p.Character:FindFirstChild("Head") and p.Character.Humanoid.Health > 0 then
                if not activeHealthGuis[p.Name] then
                    local bGui = Instance.new("BillboardGui", ScreenGui); bGui.Size = UDim2.new(0, 200, 0, 50); bGui.StudsOffset = Vector3.new(0, 3.5, 0); bGui.AlwaysOnTop = true; bGui.Adornee = p.Character.Head

                end
                activeHealthGuis[p.Name].TextLabel.Text = "❤️ "..math.floor(p.Character.Humanoid.Health)
            end
        end

end)
