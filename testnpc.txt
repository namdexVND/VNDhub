local Players = game:GetService("Players")
local VirtualUser = game:GetService("VirtualUser")
local player = Players.LocalPlayer
local Workspace = game:GetService("Workspace")

-- Tọa độ Boss cố định
local FIXED_BOSS_POS = Vector3.new(-2790, 239, -1762)

_G.Settings = {
    TargetName = "NPC2",
    DashDist = 18.2,
    Speed = 0.05
}
_G.AutoFarmLevel = false

-- [ TẠO GIAO DIỆN ]
local ScreenGui = Instance.new("ScreenGui", player:WaitForChild("PlayerGui"))
local Main = Instance.new("Frame", ScreenGui)
Main.Size = UDim2.new(0, 220, 0, 200); Main.Position = UDim2.new(0.5, -110, 0.5, -100)
Main.BackgroundColor3 = Color3.fromRGB(30, 30, 45); Main.Active = true; Main.Draggable = true
Instance.new("UICorner", Main).CornerRadius = UDim.new(0, 10)

local function createInput(name, pos, default)
    local box = Instance.new("TextBox", Main)
    box.Size = UDim2.new(0.8, 0, 0, 30); box.Position = pos
    box.PlaceholderText = name .. " (" .. default .. ")"
    box.BackgroundColor3 = Color3.fromRGB(50, 50, 70); box.TextColor3 = Color3.new(1,1,1)
    box.Text = tostring(default)
    return box
end

local distBox = createInput("Khoảng cách", UDim2.new(0.1, 0, 0.15, 0), _G.Settings.DashDist)
local speedBox = createInput("Tốc độ", UDim2.new(0.1, 0, 0.4, 0), _G.Settings.Speed)

local StartBtn = Instance.new("TextButton", Main)
StartBtn.Size = UDim2.new(0.8, 0, 0, 40); StartBtn.Position = UDim2.new(0.1, 0, 0.7, 0)
StartBtn.Text = "START FARM"; StartBtn.BackgroundColor3 = Color3.fromRGB(40, 200, 100)
StartBtn.Parent = Main

StartBtn.MouseButton1Click:Connect(function()
    _G.AutoFarmLevel = not _G.AutoFarmLevel
    if tonumber(distBox.Text) then _G.Settings.DashDist = tonumber(distBox.Text) end
    if tonumber(speedBox.Text) then _G.Settings.Speed = tonumber(speedBox.Text) end
    
    StartBtn.Text = _G.AutoFarmLevel and "STOP FARM" or "START FARM"
    StartBtn.BackgroundColor3 = _G.AutoFarmLevel and Color3.fromRGB(200, 40, 40) or Color3.fromRGB(40, 200, 100)
end)

-- [ LOGIC CHÍNH - TỐI ƯU CLICK & TELE ]
local lastDash = 0
task.spawn(function()
    while true do
        task.wait() 
        if _G.AutoFarmLevel and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            pcall(function()
                local hrp = player.Character.HumanoidRootPart
                local target = nil
                
                -- 1. Tìm Boss và ép nó về tọa độ mục tiêu
                for _, obj in pairs(Workspace:GetDescendants()) do
                    if obj.Name == _G.Settings.TargetName and obj:FindFirstChild("HumanoidRootPart") and obj:FindFirstChild("Humanoid") then
                        if obj.Humanoid.Health > 0 then
                            target = obj.HumanoidRootPart
                            target.CFrame = CFrame.new(FIXED_BOSS_POS)
                            obj.Humanoid.WalkSpeed = 0
                            break 
                        end
                    end
                end

                -- 2. Tele đón đầu và Click "sạch"
                if target and (tick() - lastDash > _G.Settings.Speed) then
                    lastDash = tick()
                    
                    local offsets = {
                        Vector3.new(_G.Settings.DashDist, 0, 0),
                        Vector3.new(-_G.Settings.DashDist, 0, 0),
                        Vector3.new(0, 0, _G.Settings.DashDist),
                        Vector3.new(0, 0, -_G.Settings.DashDist)
                    }
                    
                    local randomOffset = offsets[math.random(1, #offsets)]
                    hrp.CFrame = CFrame.lookAt(FIXED_BOSS_POS + randomOffset, FIXED_BOSS_POS)
                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                    
                    -- FIX CLICK: Sử dụng Tool nếu có, nếu không click góc 0,0 để né Menu
                    local tool = player.Character:FindFirstChildOfClass("Tool")
                    if tool then
                        tool:Activate()
                    else
                        VirtualUser:CaptureController()
                        VirtualUser:Button1Down(Vector2.new(0, 0))
                        task.delay(0.02, function() VirtualUser:Button1Up(Vector2.new(0, 0)) end)
                    end
                end
            end)
        end
    end
end)
