-- AutoHeal Toggle UI
-- Tự động dùng Băng Gạc khi máu bị mất

local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- ============================================================
-- CẤU HÌNH
-- ============================================================
local args = {
    [1] = "B\196\131ng G\225\186\161c"
}

-- ============================================================
-- TẠO GUI
-- ============================================================

screenGui.ResetOnSpawn = false
screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

-- Frame chính
local mainFrame = Instance.new("Frame")
mainFrame.Name = "MainFrame"
mainFrame.Size = UDim2.new(0, 200, 0, 60)
mainFrame.Position = UDim2.new(0, 20, 0.5, -30)
mainFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
mainFrame.BorderSizePixel = 0
mainFrame.Active = true
mainFrame.Draggable = true
mainFrame.Parent = screenGui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 12)
corner.Parent = mainFrame

local stroke = Instance.new("UIStroke")
stroke.Color = Color3.fromRGB(60, 60, 80)
stroke.Thickness = 1

-- Icon trái tim
local iconLabel = Instance.new("TextLabel")
iconLabel.Name = "Icon"
iconLabel.Size = UDim2.new(0, 30, 1, 0)

local titleLabel = Instance.new("TextLabel")

titleLabel.Size = UDim2.new(0, 100, 0.5, 0)

titleLabel.TextScaled = true

titleLabel.TextXAlignment = Enum.TextXAlignment.Left

-- Nhãn trạng thái
local statusLabel = Instance.new("TextLabel")
statusLabel.Name = "Status"
statusLabel.Size = UDim2.new(0, 100, 0.4, 0)
statusLabel.Position = UDim2.new(0, 46, 0.52, 0)
statusLabel.BackgroundTransparency = 1
statusLabel.Text = "OFF"
statusLabel.TextColor3 = Color3.fromRGB(140, 140, 160)
statusLabel.TextScaled = true
statusLabel.Font = Enum.Font.Gotham

statusLabel.Parent = mainFrame

-- Nút toggle
local toggleButton = Instance.new("Frame")
toggleButton.Name = "Toggle"
toggleButton.Size = UDim2.new(0, 44, 0, 24)
toggleButton.Position = UDim2.new(1, -54, 0.5, -12)
toggleButton.BackgroundColor3 = Color3.fromRGB(50, 50, 65)
toggleButton.BorderSizePixel = 0
toggleButton.Parent = mainFrame

local toggleCorner = Instance.new("UICorner")

toggleCorner.Parent = toggleButton

local toggleKnob = Instance.new("Frame")
toggleKnob.Name = "Knob"
toggleKnob.Size = UDim2.new(0, 18, 0, 18)
toggleKnob.Position = UDim2.new(0, 3, 0.5, -9)
toggleKnob.BackgroundColor3 = Color3.fromRGB(150, 150, 170)
toggleKnob.BorderSizePixel = 0
toggleKnob.Parent = toggleButton

local knobCorner = Instance.new("UICorner")
knobCorner.CornerRadius = UDim.new(1, 0)
knobCorner.Parent = toggleKnob

-- Vùng click trong suốt phủ lên toggle
local clickArea = Instance.new("TextButton")
clickArea.Size = UDim2.new(1, 0, 1, 0)
clickArea.BackgroundTransparency = 1
clickArea.Text = ""
clickArea.Parent = toggleButton

-- ============================================================
-- LOGIC TOGGLE
-- ============================================================
local isEnabled = false
local healThread = nil

local tweenInfo = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local function setToggle(state)
    isEnabled = state

    if state then
        -- Trạng thái ON
        TweenService:Create(toggleButton, tweenInfo, {
            BackgroundColor3 = Color3.fromRGB(60, 200, 100)
        }):Play()
        TweenService:Create(toggleKnob, tweenInfo, {
            Position = UDim2.new(0, 23, 0.5, -9),
            BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        }):Play()
        TweenService:Create(stroke, tweenInfo, {
            Color = Color3.fromRGB(60, 200, 100)
        }):Play()
        statusLabel.Text = "ON"
        statusLabel.TextColor3 = Color3.fromRGB(60, 200, 100)
    else
        -- Trạng thái OFF
        TweenService:Create(toggleButton, tweenInfo, {
            BackgroundColor3 = Color3.fromRGB(50, 50, 65)
        }):Play()
        TweenService:Create(toggleKnob, tweenInfo, {
            Position = UDim2.new(0, 3, 0.5, -9),
            BackgroundColor3 = Color3.fromRGB(150, 150, 170)
        }):Play()
        TweenService:Create(stroke, tweenInfo, {
            Color = Color3.fromRGB(60, 60, 80)
        }):Play()
        statusLabel.Text = "OFF"
        statusLabel.TextColor3 = Color3.fromRGB(140, 140, 160)
    end
end

local function startHealLoop()
    if healThread then
        task.cancel(healThread)
        healThread = nil
    end

    healThread = task.spawn(function()
        local consumableService = game:GetService("ReplicatedStorage")
            :WaitForChild("Packages")
            :WaitForChild("Knit")
            :WaitForChild("Services")
            :WaitForChild("ConsumableService")
            :WaitForChild("RF")
            :WaitForChild("UseConsumable")

        while isEnabled do
            task.wait(0.5)
            if not isEnabled then break end

            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")

            if hum and hum.Health > 0 and hum.Health < hum.MaxHealth then
                if consumableService then
                    consumableService:InvokeServer(unpack(args))
                end
            end
        end
    end)
end

local function stopHealLoop()
    if healThread then
        task.cancel(healThread)
        healThread = nil
    end
end

-- Xử lý click toggle
clickArea.MouseButton1Click:Connect(function()
    local newState = not isEnabled
    setToggle(newState)

    if newState then
        startHealLoop()
    else
        stopHealLoop()
    end
end)

-- ============================================================
-- KHỞI ĐỘNG
-- ============================================================
setToggle(false)
print("[AutoHeal UI] Đã tải xong. Nhấn toggle để bật/tắt tự động hồi máu.")
