local function getFixedKeyFor8Hours()
    local totalHours = math.floor(os.time() / 3600)
    local period = math.floor(totalHours / 8)
    math.randomseed(period)
    local randomString = ""
    for i = 1, 12 do randomString = randomString .. tostring(math.random(0, 9)) end
    return "VND_" .. randomString

-- Logic chạy ngầm của Farm Boss (Đã fix Aim thẳng Boss không giật Camera)

local VirtualUser = game:GetService("VirtualUser")
local player = game.Players.LocalPlayer
local lastDash = 0
local hasFollowed = false 
local spawnTime = 0 -- Biến lưu thời điểm phát hiện boss
local targetPos = Vector3.new(-5009, 182, -108)

task.spawn(function()
    while task.wait() do
        if _G.AutoFarmBoss then
            pcall(function()
                local char = player.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                local isSitting = (hum and hum.Sit) or (hrp and hrp:FindFirstChild("SeatWeld"))

                if hrp and not isSitting then
                    -- 1. Noclip
                    for _, part in pairs(char:GetChildren()) do
                        if part:IsA("BasePart") then part.CanCollide = false end
                    end

                    -- Tìm Boss
                    local targetHead = nil
                    for _, obj in pairs(workspace:GetDescendants()) do
                        if obj.Name == _G.BossTargetName and obj:FindFirstChild("Head") and obj:FindFirstChild("Humanoid") then
                            if obj.Humanoid.Health > 0 then
                                targetHead = obj.Head
                                break
                            end
                        end
                    end

                    -- 2. Logic dịch chuyển
                    if targetHead then
                        if not hasFollowed then
                            -- Nếu mới thấy boss lần đầu, ghi lại thời gian
                            if spawnTime == 0 then
                                spawnTime = tick()
                            end

                            -- Đợi đủ 3 giây kể từ khi thấy boss mới bắt đầu di chuyển
                            if tick() - spawnTime >= 3 then
                                hrp.CFrame = CFrame.new(targetHead.Position + Vector3.new(37, 6, 0))
                                task.wait(2)
                                hasFollowed = true
                            end
                        else
                            -- Sau khi đã bay đến đầu boss lần đầu, duy trì ở targetPos
                            local lookAtPos = Vector3.new(targetHead.Position.X, hrp.Position.Y, targetHead.Position.Z)
                            hrp.CFrame = CFrame.new(targetPos, lookAtPos)
                        end
                    else
                        -- Reset khi không thấy boss
                        hasFollowed = false
                        spawnTime = 0
                        hrp.CFrame = CFrame.new(targetPos)
                    end
                    
                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                    
                    -- 3. Tấn công
                    if targetHead and hasFollowed and (tick() - lastDash > _G.AttackSpeed) then
                        lastDash = tick()
                        local tool = char:FindFirstChildOfClass("Tool")
                        if tool then
                            tool:Activate()
                        else
                            VirtualUser:CaptureController()
                            VirtualUser:Button1Down(Vector2.new(0, 0))
                            task.delay(0.02, function() VirtualUser:Button1Up(Vector2.new(0, 0)) end)
                        end
                    end
                end
            end)
        else
            hasFollowed = false
            spawnTime = 0
        end
    end
end)

local function getBoss2()
    local gh2 = workspace:FindFirstChild("GiangHo2")

SetupArenaCamera = function(enabled)
    if enabled then
        if cameraConnection then cameraConnection:Disconnect() end
        if cameraPart then cameraPart:Destroy() end
        
        cameraPart = Instance.new("Part")
        cameraPart.Name = "ArenaCameraAnchor"
        cameraPart.Size = Vector3.new(1, 1, 1)
        cameraPart.Transparency = 1
        cameraPart.CanCollide = false
        cameraPart.Anchored = true
        cameraPart.Parent = workspace
        
        workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable
        currentAngle = 0
        
        cameraConnection = RunService.RenderStepped:Connect(function(deltaTime)
            currentAngle = currentAngle + (OrbitSpeed * deltaTime)
            local offsetX = math.cos(currentAngle) * ArenaRadius
            local offsetZ = math.sin(currentAngle) * ArenaRadius
            local currentCenter = ArenaCenter
            local liveBoss = getBoss2()
            if liveBoss and liveBoss:FindFirstChild("HumanoidRootPart") then
                currentCenter = liveBoss.HumanoidRootPart.Position
            end
            local cameraPos = Vector3.new(currentCenter.X + offsetX, currentCenter.Y + CameraHeight, currentCenter.Z + offsetZ)
            cameraPart.CFrame = CFrame.new(cameraPos, currentCenter)
            workspace.CurrentCamera.CFrame = cameraPart.CFrame
        end)
    else
        if cameraConnection then cameraConnection:Disconnect() end
        if cameraPart then cameraPart:Destroy() end
        workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
        if player.Character and player.Character:FindFirstChild("Humanoid") then
            workspace.CurrentCamera.CameraSubject = player.Character.Humanoid
        end
    end
end

local Lighting = game:GetService("Lighting")

local function activateFPSBoost()
    pcall(function()
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 9e9
        Lighting.Brightness = 0

        if Terrain then
            Terrain.WaterWaveSize = 0
            Terrain.WaterWaveSpeed = 0
            Terrain.WaterReflectance = 0
            Terrain.WaterTransparency = 1
        end

        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    end)

    for _,v in ipairs(workspace:GetDescendants()) do
        pcall(function()

            if v:IsA("ParticleEmitter")
            or v:IsA("Trail")
            or v:IsA("Beam")
            or v:IsA("Smoke")
            or v:IsA("Fire")
            or v:IsA("Sparkles") then
                v.Enabled = false
            end

            if v:IsA("BasePart") then
                v.CastShadow = false
                v.Material = Enum.Material.SmoothPlastic
                v.Reflectance = 0
            end

        end)
    end
end

local function deactivateFPSBoost()
    pcall(function()
        Lighting.GlobalShadows = true
        Lighting.Brightness = 2
    end)
end

local function toggleGraphicReduction(percent, enabled)

    if not enabled then
        return
    end

    local parts = {}

    for _,v in ipairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") then
            table.insert(parts,v)
        end
    end

    local amount = math.floor(#parts * (percent / 100))

    for i = 1, amount do
        local obj = parts[i]

        pcall(function()

            if not savedGraphics[obj] then
                savedGraphics[obj] = {
                    Transparency = obj.Transparency,
                    Material = obj.Material
                }
            end

            obj.Transparency = 1

        end)
    end
end

local soLanGau, soLanGa, tongSoPhien, lichSuCau = 0, 0, 0, {}
local processedSessions = {} 

local function layChuoiHienTai()
	if #lichSuCau == 0 then return 0, "" end
	local loaiCuoi = lichSuCau[#lichSuCau]
	local count = 0
	for i = #lichSuCau, 1, -1 do
		if lichSuCau[i] == loaiCuoi then count = count + 1 else break end
	end
	return count, loaiCuoi
end

-- =======================================================================
-- TẦNG 1: PHÂN TÍCH TOÁN HỌC CHUẨN XU HƯỚNG THEO CÁC DÒNG CẦU ĐẶC THÙ
-- =======================================================================
local function tinhToanPhanTich()
	local n = #lichSuCau
	if n < 2 then return '<font color="#FFE600">Chờ dữ liệu...</font>', "0%" end

	local chuoiCount, loaiChuoi = layChuoiHienTai()
	local gauText = '<font color="#33FF33">🐻 GẤU</font>'
	local gaText = '<font color="#FF3333">🐔 GÀ</font>'
	
	local nguocIcon = (loaiChuoi == "Gau") and gaText or gauText
	local xuatIcon = (loaiChuoi == "Gau") and gauText or gaText

	-- 1. ƯU TIÊN BỆT MẠNH: Xuất hiện từ 3 ván cùng loại trở lên -> Đuổi bệt liên tục
	if chuoiCount >= 3 then 
		return xuatIcon .. " [Cầu Bệt]", "80%" 
	end

	-- Nhận diện các thế cầu khi có ít nhất 3 phiên dữ liệu
	if n >= 3 then
		local c1, c2, c3 = lichSuCau[n], lichSuCau[n-1], lichSuCau[n-2]
		
		-- CẦU 1-1 CHUẨN XEN KẼ (Ví dụ: Gà - Gấu - Gà) -> Ván tiếp ra đối ứng (Gấu) với tỷ lệ chuẩn 50%
		if c1 ~= c2 and c2 ~= c3 then
			return nguocIcon .. " [Cầu 1-1]", "50%"
		end
	end

	-- Nhận diện các mẫu cầu khi có ít nhất 4 phiên dữ liệu trở lên
	if n >= 4 then
		local c1, c2, c3, c4 = lichSuCau[n], lichSuCau[n-1], lichSuCau[n-2], lichSuCau[n-3]
		
		-- CẦU 2-2 CHUẨN (Ví dụ: Gấu - Gấu - Gà - Gà)
		if c1 == c2 and c2 ~= c3 and c3 == c4 then 
			return nguocIcon .. " [Cầu 2-2]", "75%" 
		end

		-- CẦU 1-2 (Ví dụ: Gà - Gấu - Gấu - Gà) -> Đang đi cấu trúc đôi, ván tiếp ra trùng ván c1
		if c1 ~= c2 and c2 == c3 and c3 ~= c4 then
			return xuatIcon .. " [Cầu 1-2]", "70%"
		end

		-- CẦU Đối xứng hoàn chỉnh
		if c1 ~= c2 and c2 == c3 and c3 == c4 then
			return xuatIcon .. " [Đối Xứng]", "70%"
		end
	end

	-- Thuật toán Quét mật độ (Cầu Nghiêng) dựa trên 10 ván gần nhất
	local gGau, gGa = 0, 0
	for i = math.max(1, n-9), n do
		if lichSuCau[i] == "Gau" then gGau = gGau + 1 else gGa = gGa + 1 end
	end
	if math.abs(gGau - gGa) >= 3 then
		return (gGau > gGa and gauText or gaText) .. " [Cầu Nghiêng]", "65%"
	end

	return nguocIcon .. " [Cầu Nhảy]", "50%"
end

-- =======================================================================
-- TẦNG 2: CON AI CẢNH BÁO ĐỘC LẬP (CHỈ ĐƯA RA CẢNH BÁO BIẾN ĐỘNG TRẠNG THÁI)
-- =======================================================================
local function AI_CanhBaoDocLap()
	local chuoiCount, loaiChuoi = layChuoiHienTai()
	if chuoiCount == 0 then return '<font color="#888888">Bình thường</font>' end

	if chuoiCount >= 8 then
		return '<font color="#FF3333"><b>95% SẬP CẦU DỨT ĐUÔI!</b></font>'
	elseif chuoiCount >= 6 then
		return '<font color="#FFA500"><b>70% KHẢ NĂNG BẺ CẦU</b></font>'
	elseif chuoiCount == 5 then
		return '<font color="#FFFF00"><b>50% ĐẢO CHIỀU CỰC MẠNH</b></font>'
	elseif chuoiCount == 4 then
		return '<font color="#00FFFF">Chuỗi bệt tăng dần (Chú ý)</font>'
	end

	return '<font color="#00FF66">An toàn</font>'
end

local function capNhatBang()
	local phanTramGau = tongSoPhien > 0 and math.round((soLanGau / tongSoPhien) * 100) or 0
	local phanTramGa = tongSoPhien > 0 and math.round((soLanGa / tongSoPhien) * 100) or 0
	
	local duDoanPhanTich, doTinCay = tinhToanPhanTich()
	local thongTinCanhBao = AI_CanhBaoDocLap()
	
	local targetText = InputPhien.Text ~= "" and InputPhien.Text or "Trống"
	local chuoiIcon = ""
	for i = math.max(1, #lichSuCau - 7), #lichSuCau do chuoiIcon = chuoiIcon .. (lichSuCau[i] == "Gau" and "🐻" or "🐔") end
	if chuoiIcon == "" then chuoiIcon = "..." end

	TextDisplay.Text = string.format(
		"🎯 <b>Mục tiêu:</b> <font color='#00FFCC'>#%s</font>\n" ..
		"🐔 <b>Gà:</b> %d (%d%%) | 🐻 <b>Gấu:</b> %d (%d%%)\n" ..
		"📊 <b>Tổng số ván:</b> %d\n" ..
		"🕒 <b>Chuỗi:</b> %s\n" ..
		"-----------------------------------------\n" ..
		"📊 <b>Phân tích cầu:</b> %s (%s)\n" ..
		"⚠️ <b>AI Cảnh báo:</b> %s",
		targetText, soLanGa, phanTramGa, soLanGau, phanTramGau, tongSoPhien, chuoiIcon, duDoanPhanTich, doTinCay, thongTinCanhBao
	)
end

local function ghiNhan(ketQua)
	tongSoPhien += 1
	if ketQua == "Gau" then soLanGau += 1; table.insert(lichSuCau, "Gau") else soLanGa += 1; table.insert(lichSuCau, "Ga") end
	
	local num = tonumber(InputPhien.Text)
	if num then InputPhien.Text = tostring(num + 1) end
	capNhatBang()
end

                and server.id ~= game.JobId then

                    TeleportService:TeleportToPlaceInstance(
                        game.PlaceId,
                        server.id,
                        player
                    )

                    break
                end
            end

        end)

    end
})
