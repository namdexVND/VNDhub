local plr = game:GetService("Players").LocalPlayer
local autoSitEnabled = true
local scanRadius = 20 -- Bán kính quét 20 studs

-- 1. Giao diện (GUI)
local gui = Instance.new("ScreenGui", plr:WaitForChild("PlayerGui"))
local toggleBtn = Instance.new("TextButton", gui)
toggleBtn.Size = UDim2.new(0, 110, 0, 30)
toggleBtn.Position = UDim2.new(0, 40, 0, 390)
toggleBtn.BackgroundColor3 = Color3.fromRGB(15, 10, 25)
toggleBtn.Text = "🟢 TỰ LỤM: BẬT"
toggleBtn.Font = Enum.Font.ArialBold 
toggleBtn.TextSize = 13
toggleBtn.TextColor3 = Color3.fromRGB(180, 70, 255)
toggleBtn.Active = true 
toggleBtn.Draggable = true

local btnStroke = Instance.new("UIStroke", toggleBtn)
btnStroke.Color = Color3.fromRGB(130, 50, 200)
btnStroke.Thickness = 1.5
Instance.new("UICorner", toggleBtn).CornerRadius = UDim.new(0, 6)

toggleBtn.MouseButton1Click:Connect(function()
    autoSitEnabled = not autoSitEnabled
    toggleBtn.Text = autoSitEnabled and "🟢 TỰ LỤM: BẬT" or "🔴 TỰ LỤM: TẮT"
end)

-- 2. Vòng lặp quét đối tượng
task.spawn(function()
    while true do
        task.wait(0.5)
        if autoSitEnabled then
            local char = plr.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            
            if hrp then
                -- Quét xung quanh người chơi trong bán kính 20 studs
                local params = OverlapParams.new()
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = {char}
                
                local objectsNearby = workspace:GetPartBoundsInRadius(hrp.Position, scanRadius, params)
                
                for _, part in pairs(objectsNearby) do
                    -- Tìm NPC tên "CityNPC" HOẶC ProximityPrompt có tên "lụm" (hoặc liên quan)
                    local parent = part.Parent
                    
                    -- Kiểm tra NPC
                    if parent.Name == "CityNPC" or part.Name == "CityNPC" then
                        -- Xử lý tương tác với NPC ở đây nếu cần
                    end
                    
                    -- Tìm ProximityPrompt để lụm đồ
                    local prompt = part:FindFirstChildOfClass("ProximityPrompt") or parent:FindFirstChildOfClass("ProximityPrompt")
                    if prompt then
                        -- Kiểm tra nếu tên Prompt chứa chữ "lụm" (không phân biệt hoa thường)
                        if string.find(string.lower(prompt.ActionText), "lụm") or string.find(string.lower(prompt.ObjectText), "lụm") then
                            fireproximityprompt(prompt)
                        end
                    end
                end
            end
        end
    end
end)
