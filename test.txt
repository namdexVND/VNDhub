--// VND HUB - FULL CẤU HÌNH - SIÊU TỐC
local RunService = game:GetService("RunService")
local VirtualUser = game:GetService("VirtualUser")
local LocalPlayer = game:GetService("Players").LocalPlayer

-- [ CẤU HÌNH MẶC ĐỊNH CHO CityNPC ]
local settings = {
    Enabled = false,
    TargetName = "CityNPC", -- Đã set đúng tên bạn yêu cầu
    Radius = 6,             
    OrbitSpeed = 30,        
    DashSpeed = 0.2,        
    Height = 4              
}

-- [ UI ĐIỀU KHIỂN ]
local ScreenGui = Instance.new("ScreenGui", LocalPlayer:WaitForChild("PlayerGui"))
local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 250, 0, 350)
MainFrame.Position = UDim2.new(0.5, -125, 0.4, -175)
MainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 30)
MainFrame.Draggable = true
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 10)
Instance.new("UIStroke", MainFrame).Color = Color3.fromRGB(150, 80, 250)

local function createSlider(name, varName, min, max)
    local label = Instance.new("TextLabel", MainFrame)
    label.Size = UDim2.new(0.9, 0, 0, 30)
    label.Position = UDim2.new(0.05, 0, 0, #MainFrame:GetChildren() * 40)
    label.Text = name .. ": " .. settings[varName]
    label.TextColor3 = Color3.new(1,1,1)
    label.BackgroundTransparency = 1
    
    local input = Instance.new("TextBox", label)
    input.Size = UDim2.new(0.3, 0, 1, 0)
    input.Position = UDim2.new(1, 0, 0, 0)
    input.PlaceholderText = "Sửa"
    input.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
    input.FocusLost:Connect(function()
        local val = tonumber(input.Text)
        if val then settings[varName] = math.clamp(val, min, max); label.Text = name .. ": " .. settings[varName] end
    end)
end

createSlider("Khoảng cách", "Radius", 2, 20)
createSlider("Tốc độ xoay", "OrbitSpeed", 5, 100)
createSlider("Tốc độ lao", "DashSpeed", 0.05, 1)
createSlider("Độ cao", "Height", 0, 15)

local ToggleBtn = Instance.new("TextButton", MainFrame)
ToggleBtn.Size = UDim2.new(0.9, 0, 0, 50)
ToggleBtn.Position = UDim2.new(0.05, 0, 0.8, 0)
ToggleBtn.Text = "START FARM"
ToggleBtn.BackgroundColor3 = Color3.fromRGB(120, 60, 200)
ToggleBtn.TextColor3 = Color3.new(1,1,1)
ToggleBtn.MouseButton1Click:Connect(function()
    settings.Enabled = not settings.Enabled
    ToggleBtn.Text = settings.Enabled and "STOP FARM" or "START FARM"
end)

-- [ LOGIC FARM CHÍNH ]
RunService.Heartbeat:Connect(function()
    if not settings.Enabled then return end
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    
    local hrp = char.HumanoidRootPart
    local target = nil
    
    -- Tự tìm CityNPC
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") then
            target = obj.HumanoidRootPart
            break
        end
    end

    if target then
        local angle = tick() * settings.OrbitSpeed
        local pos = target.Position + Vector3.new(math.cos(angle) * settings.Radius, settings.Height, math.sin(angle) * settings.Radius)
        
        hrp.CFrame = hrp.CFrame:Lerp(CFrame.new(pos, target.Position), settings.DashSpeed)
        
        VirtualUser:ClickButton1(Vector2.new())
    end
end)
