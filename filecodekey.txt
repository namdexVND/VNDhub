local HttpService = game:GetService("HttpService")
local FILE_NAME = "VND_KEY_CACHE.txt"
local API_TOKEN = "695f7e6a085a446e0a3e7c89"
local WEB_URL = "https://vndhubfreekey.blogspot.com/2026/05/he-thong-xac-thuc-key-function.html"

-- 1. Hàm tạo mã Key cố định 8 giờ
local function getFixedKeyFor8Hours()
    local totalHours = math.floor(os.time() / 3600)
    local period = math.floor(totalHours / 8) -- Chia cho 8 để tạo chu kỳ 8 tiếng
    math.randomseed(period)
    local randomString = ""
    for i = 1, 12 do randomString = randomString .. tostring(math.random(0, 9)) end
    return "VND_" .. randomString
end

local MyCurrentKey = getFixedKeyFor8Hours()
local TargetUrl = WEB_URL .. MyCurrentKey
local ShortenedLink = "Đang lấy link..."

-- 2. Luồng lấy link rút gọn từ API
task.spawn(function()
    local encodedUrl = HttpService:UrlEncode(TargetUrl)
    local api_endpoint = "https://link4m.co/api-shorten/v2?api=" .. API_TOKEN .. "&url=" .. encodedUrl
    local success, response = pcall(function() return game:HttpGet(api_endpoint) end)
    if success then
        local data = HttpService:JSONDecode(response)
        ShortenedLink = (data and data.status == "success") and data.shortenedUrl or "Lỗi API"
    end
end)

-- 3. Giao diện (GUI)
local ScreenGui = Instance.new("ScreenGui", game:GetService("CoreGui"))
local Frame = Instance.new("Frame", ScreenGui)
Frame.Size = UDim2.new(0, 320, 0, 170)
Frame.Position = UDim2.new(0.5, -160, 0.5, -85)
Frame.BackgroundColor3 = Color3.fromRGB(20, 16, 36)
Instance.new("UICorner", Frame).CornerRadius = UDim.new(0, 12)

local TextBox = Instance.new("TextBox", Frame)
TextBox.Size = UDim2.new(0.9, 0, 0, 38)
TextBox.Position = UDim2.new(0.05, 0, 0.28, 0)
TextBox.PlaceholderText = "Dán Key vào đây..."

local CheckButton = Instance.new("TextButton", Frame)
CheckButton.Text = "Kiểm Tra Key ✅"
CheckButton.Size = UDim2.new(0.42, 0, 0, 36)
CheckButton.Position = UDim2.new(0.05, 0, 0.62, 0)
CheckButton.BackgroundColor3 = Color3.fromRGB(16, 185, 129)

local GetKeyButton = Instance.new("TextButton", Frame)
GetKeyButton.Text = "Lấy Link Key 🌐"
GetKeyButton.Size = UDim2.new(0.42, 0, 0, 36)
GetKeyButton.Position = UDim2.new(0.53, 0, 0.62, 0)
GetKeyButton.BackgroundColor3 = Color3.fromRGB(168, 85, 247)

GetKeyButton.MouseButton1Click:Connect(function()
    setclipboard(ShortenedLink)
    game.StarterGui:SetCore("SendNotification", {Title="VND HUB", Text="Link đã copy! Hãy chờ 80s trên web để lấy Key."})
end)

-- 4. Logic xác thực 8 tiếng
local keyVerified = false
local function checkSavedKey()
    if isfile and isfile(FILE_NAME) then
        if readfile(FILE_NAME) == MyCurrentKey then return true end
    end
    return false
end

if checkSavedKey() then
    keyVerified = true
    ScreenGui:Destroy()
else
    CheckButton.MouseButton1Click:Connect(function()
        if TextBox.Text == MyCurrentKey then
            keyVerified = true
            writefile(FILE_NAME, TextBox.Text)
            ScreenGui:Destroy()
        end
    end)
end

repeat task.wait() until keyVerified == true
