local HttpService = game:GetService("HttpService")

-- ⚙️ CẤU HÌNH THÔNG TIN CỦA BẠN:
local API_TOKEN = "695f7e6a085a446e0a3e7c89" -- Dán mã Token API Link4M của bạn
local WEB_URL = "https://vndhubfreekey.blogspot.com/2026/05/he-thong-xac-thuc-key-function.html?ma=" -- Đường dẫn trang web của bạn

-- Hàm tạo chuỗi SỐ ngẫu nhiên hoàn toàn (Không chữ cái)
local function generateRandomNumericKey()
    local randomString = ""
    math.randomseed(os.time() + math.random(1, 1000)) -- Tăng độ nhiễu seed để không bị trùng trùng lặp
    
    -- Vòng lặp chạy 10 lần để lấy ra 10 chữ số ngẫu nhiên từ 0 đến 9
    for i = 1, 10 do
        local num = math.random(0, 9)
        randomString = randomString .. tostring(num)
    end
    return "VND_" .. randomString
end

-- Khởi tạo mã Key ngẫu nhiên cho phiên chạy này
local MyCurrentKey = generateRandomNumericKey()
local TargetUrl = WEB_URL .. "?ma=" .. MyCurrentKey
local ShortenedLink = "Đang kết nối hệ thống lấy link..."

-- Luồng xử lý rút gọn link ngầm bằng API Link4M giống cơ chế requests bên Python
task.spawn(function()
    -- Thực hiện Encode URL để tránh lỗi ký tự đặc biệt khi gửi API
    local encodedUrl = HttpService:UrlEncode(TargetUrl)
    local api_endpoint = "https://link4m.co/api-shorten/v2?api=" .. API_TOKEN .. "&url=" .. encodedUrl
    
    local success, response = pcall(function()
        return game:HttpGet(api_endpoint)
    end)
    
    if success then
        local data = HttpService:JSONDecode(response)
        -- Phân tích cú pháp JSON trả về từ hệ thống Link4M
        if data and data.status == "success" and data.shortenedUrl then
            ShortenedLink = data.shortenedUrl
        elseif data and data.message then
            ShortenedLink = "Lỗi hệ thống: " .. tostring(data.message)
        else
            ShortenedLink = "Lỗi cấu hình: API Token không hợp lệ!"
        end
    else
        ShortenedLink = "Lỗi kết nối: Không thể gọi dữ liệu từ Link4M!"
    end
end)
