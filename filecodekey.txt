local HttpService = game:GetService("HttpService")
local FILE_NAME = "VND_KEY_CACHE.txt"
local API_TOKEN = "695f7e6a085a446e0a3e7c89"
local WEB_URL = "https://vndhubfreekey.blogspot.com/2026/05/he-thong-xac-thuc-key-function.html"

local function getFixedKeyFor8Hours()
    local totalHours = math.floor(os.time() / 3600)
    local period = math.floor(totalHours / 8)
    math.randomseed(period)
    local randomString = ""
    for i = 1, 12 do randomString = randomString .. tostring(math.random(0, 9)) end
    return "VND_" .. randomString
end

local MyCurrentKey = getFixedKeyFor8Hours()
-- Tạo link với dấu ? chuẩn xác
local TargetUrl = WEB_URL .. "?ma=" .. MyCurrentKey 
local ShortenedLink = "Đang lấy link..."

task.spawn(function()
    local api_endpoint = "https://link4m.co/api-shorten/v2?api=" .. API_TOKEN .. "&url=" .. HttpService:UrlEncode(TargetUrl)
    local success, response = pcall(function() return game:HttpGet(api_endpoint) end)
    if success then
        local data = HttpService:JSONDecode(response)
        ShortenedLink = (data and data.status == "success") and data.shortenedUrl or TargetUrl
    end
end)

local keyVerified = false
if isfile and isfile(FILE_NAME) and readfile(FILE_NAME) == MyCurrentKey then
    keyVerified = true
    ScreenGui:Destroy()
else

        if TextBox.Text == MyCurrentKey then
            keyVerified = true
            writefile(FILE_NAME, TextBox.Text)

    end)
