local RunService = game:GetService("RunService")
local VirtualUser = game:GetService("VirtualUser")
local LocalPlayer = game:GetService("Players").LocalPlayer
local Workspace = game:GetService("Workspace")

local config = {
    Enabled = false,
    TargetName = "CityNPC",
    DashDist = 15, -- Mặc định
    Speed = 0.2    -- Mặc định
}

-- [ TẠO GIAO DIỆN CÓ KHUNG NHẬP LIỆU ]
local ScreenGui = Instance.new("ScreenGui", LocalPlayer:WaitForChild("PlayerGui"))
local Main = Instance.new("Frame", ScreenGui)
Main.Size = UDim2.new(0, 250, 0, 250); Main.Position = UDim2.new(0.5, -125, 0.5, -125)
Main.BackgroundColor3 = Color3.fromRGB(30, 30, 45); Main.Active = true; Main.Draggable = true
Instance.new("UICorner", Main).CornerRadius = UDim.new(0, 10)

-- Hàm tạo ô nhập liệu
local function createInput(name, pos, default)
    local box = Instance.new("TextBox", Main)
    box.Size = UDim2.new(0.8, 0, 0, 30); box.Position = pos
    box.PlaceholderText = name .. " (Hiện tại: " .. default .. ")"
    box.BackgroundColor3 = Color3.fromRGB(50, 50, 70); box.TextColor3 = Color3.new(1,1,1)
    return box
end

local distBox = createInput("Khoảng cách", UDim2.new(0.1, 0, 0.15, 0), config.DashDist)
local speedBox = createInput("Tốc độ (s)", UDim2.new(0.1, 0, 0.35, 0), config.Speed)

local StartBtn = Instance.new("TextButton", Main)
StartBtn.Size = UDim2.new(0.8, 0, 0, 40); StartBtn.Position = UDim2.new(0.1, 0, 0.65, 0)
StartBtn.Text = "START FARM"; StartBtn.BackgroundColor3 = Color3.fromRGB(40, 200, 100); StartBtn.Parent = Main

StartBtn.MouseButton1Click:Connect(function()
    config.Enabled = not config.Enabled
    -- Cập nhật giá trị từ ô nhập
    if tonumber(distBox.Text) then config.DashDist = tonumber(distBox.Text) end
    if tonumber(speedBox.Text) then config.Speed = tonumber(speedBox.Text) end
    
    StartBtn.Text = config.Enabled and "STOP FARM" or "START FARM"
    StartBtn.BackgroundColor3 = config.Enabled and Color3.fromRGB(200, 40, 40) or Color3.fromRGB(40, 200, 100)
end)

-- [ LOGIC CHÍNH: DỊCH CHUYỂN TỨC THỜI ]
local lastAction = 0

RunService.Heartbeat:Connect(function()
    if not config.Enabled then return end
    
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local target = nil
    
    -- Tìm NPC
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj.Name == config.TargetName and obj:FindFirstChild("HumanoidRootPart") then
            if obj.Humanoid.Health > 0 then target = obj.HumanoidRootPart; break end
        end
    end

    if target and hrp and (tick() - lastAction > config.Speed) then
        lastAction = tick()
        
        -- DỊCH CHUYỂN TỨC THỜI (Teleport đến vị trí xung quanh Boss)
        local randomAngle = math.random() * math.pi * 2
        local offset = Vector3.new(math.cos(randomAngle), 0, math.sin(randomAngle)) * config.DashDist
        hrp.CFrame = CFrame.lookAt(target.Position + offset, target.Position)
        
        -- Tấn công
        VirtualUser:CaptureController()
        VirtualUser:Button1Down(Vector2.new(0.5, 0.5))
        task.wait(0.05)
        VirtualUser:Button1Up(Vector2.new(0.5, 0.5))
    end
end)
