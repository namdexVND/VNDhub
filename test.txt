local Players = game:GetService("Players")
local VirtualUser = game:GetService("VirtualUser")
local player = Players.LocalPlayer

local settings = {
    TargetName = "NPC2", -- Tên NPC (Phải khớp chính xác trong game)
    DashDist = 16,       -- Khoảng cách phi đến quái
    Speed = 0.06         -- Tốc độ thực hiện hành động (càng nhỏ càng nhanh)
}

local lastDash = 0

task.spawn(function()
    while true do
        task.wait() -- Chờ mỗi frame để đạt tốc độ cao nhất
        
        -- Chỉ chạy nếu công tắc AutoFarm được bật và nhân vật đã tồn tại
        if _G.AutoFarmLevel and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            pcall(function()
                local hrp = player.Character.HumanoidRootPart
                local target = nil
                
                -- Tìm NPC theo tên
                for _, obj in pairs(workspace:GetDescendants()) do
                    if obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") and obj:FindFirstChild("Humanoid") then
                        if obj.Humanoid.Health > 0 then 
                            target = obj.HumanoidRootPart
                            break 
                        end
                    end
                end

                -- Nếu tìm thấy mục tiêu và đủ thời gian chờ (Speed)
                if target and (tick() - lastDash > settings.Speed) then
                    lastDash = tick()
                    
                    -- Tính toán vị trí xung quanh quái để phi vào
                    local randomAngle = math.random() * math.pi * 2
                    local offset = Vector3.new(math.cos(randomAngle), 0, math.sin(randomAngle)) * settings.DashDist
                    
                    -- Dịch chuyển nhân vật đến vị trí đã tính toán và hướng mặt về phía quái
                    hrp.CFrame = CFrame.lookAt(target.Position + offset, target.Position)
                    
                    -- Thực hiện Auto Click giả lập
                    VirtualUser:CaptureController()
                    VirtualUser:Button1Down(Vector2.new(0.5, 0.5))
                    task.delay(0.02, function()
                        VirtualUser:Button1Up(Vector2.new(0.5, 0.5))
                    end)
                end
            end)
        end
    end
end)
