local RunService = game:GetService("RunService")
local VirtualUser = game:GetService("VirtualUser")
local LocalPlayer = game:GetService("Players").LocalPlayer
local Workspace = game:GetService("Workspace")

local settings = {
    Enabled = false,
    TargetName = "CityNPC",
    DashDistance = 15, -- Khoảng cách lướt qua quái
    AttackRange = 25   -- Khoảng cách bắt đầu farm
}

-- [ GIAO DIỆN (Giữ nguyên menu cũ của bạn) ]
-- (Phần giao diện mình đã lược bớt để bạn dễ copy, chỉ tập trung vào logic)

-- [ LOGIC FARM NÂNG CẤP ]
local lastDash = 0
local dashCooldown = 0.35 -- Tốc độ lướt (càng thấp càng nhanh)

RunService.Heartbeat:Connect(function()
    if not settings.Enabled then return end
    
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local target = nil
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj.Name == settings.TargetName and obj:FindFirstChild("HumanoidRootPart") then
            local hum = obj:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                target = obj.HumanoidRootPart
                break
            end
        end
    end

    if target then
        local dist = (hrp.Position - target.Position).Magnitude
        
        -- NẾU QUÁ XA: Tự động lao tới
        if dist > settings.AttackRange then
            hrp.CFrame = CFrame.lookAt(hrp.Position, target.Position) * CFrame.new(0, 0, -2)
        
        -- NẾU ĐỦ GẦN: Phi siêu tốc (Snap Dash)
        elseif tick() - lastDash > dashCooldown then
            lastDash = tick()
            
            -- Lướt ngẫu nhiên trái hoặc phải so với hướng nhìn quái
            local side = (math.random() > 0.5 and 1 or -1) * settings.DashDistance
            local snapPos = target.CFrame * CFrame.new(side, 0, 8) -- Lướt qua cách 8 studs
            
            hrp.CFrame = CFrame.lookAt(snapPos.Position, target.Position)
            
            -- Đánh dồn dập
            VirtualUser:CaptureController()
            VirtualUser:Button1Down(Vector2.new(0.5, 0.5))
            task.spawn(function()
                task.wait(0.05)
                VirtualUser:Button1Up(Vector2.new(0.5, 0.5))
            end)
        end
    end
end)
