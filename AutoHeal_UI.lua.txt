end

-- LOGIC KIỂM SOÁT TỐC ĐỘ CAO
-- ============================================================
local isEnabled = false
local healConnection = nil -- Sử dụng Connection thay cho Thread

local tweenInfo = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

local function setToggle(state)
    isEnabled = state
    if state then
        TweenService:Create(toggleButton, tweenInfo, {BackgroundColor3 = Color3.fromRGB(60, 200, 100)}):Play()
        TweenService:Create(toggleKnob, tweenInfo, {Position = UDim2.new(0, 23, 0.5, -9), BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()
        TweenService:Create(stroke, tweenInfo, {Color = Color3.fromRGB(60, 200, 100)}):Play()
        statusLabel.Text = "ON"
        statusLabel.TextColor3 = Color3.fromRGB(60, 200, 100)
    else
        TweenService:Create(toggleButton, tweenInfo, {BackgroundColor3 = Color3.fromRGB(50, 50, 65)}):Play()
        TweenService:Create(toggleKnob, tweenInfo, {Position = UDim2.new(0, 3, 0.5, -9), BackgroundColor3 = Color3.fromRGB(150, 150, 170)}):Play()
        TweenService:Create(stroke, tweenInfo, {Color = Color3.fromRGB(60, 60, 80)}):Play()
        statusLabel.Text = "OFF"
        statusLabel.TextColor3 = Color3.fromRGB(140, 140, 160)
    end
end

local function startHealLoop()
    if healConnection then
        healConnection:Disconnect()
        healConnection = nil
    end

    -- CHẠY SIÊU TỐC: Kích hoạt kiểm tra theo thời gian thực của khung hình máy tính (Heartbeat)
    healConnection = RunService.Heartbeat:Connect(function()
        if currentHumanoid and currentHumanoid.Health > 0 and currentHumanoid.Health < currentHumanoid.MaxHealth then
            if consumableService then
                -- TỐI ƯU TUYỆT ĐỐI: Gửi trực tiếp phương thức hệ thống, KHÔNG tạo closure function mới, KHÔNG sinh rác bộ nhớ
                task.spawn(consumableService.InvokeServer, consumableService, ITEM_NAME)
            end
        end
    end)
end

local function stopHealLoop()
    if healConnection then
        healConnection:Disconnect()
        healConnection = nil
    end
end

clickArea.MouseButton1Click:Connect(function()
    local newState = not isEnabled
    setToggle(newState)
    if newState then startHealLoop() else stopHealLoop() end
end)

setToggle(false)
print("[AutoHeal UI] Đã kích hoạt cấu trúc Ép Xung Cực Hạn - Tốc độ tối đa lý thuyết!")
